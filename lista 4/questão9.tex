\section{Questão 9}

Com base nos resultados anteriores para o experimento randômico, calcule as probabilidades condicionais:

\begin{multicols}{2}
\begin{itemize}
    \item [a)] $P_{X\mid Y} (X=0\mid 1\leq Y\leq 3)$
    \item [b)] $P_{Y\mid X} (Y=0\mid 1\leq X\leq 3)$
    \item [c)] $P_{Y\mid X} (Y=2\mid X\leq 2)$
    \item [d)] $P_{X\mid Y} (X\leq 1\mid Y\geq 2)$
    \item [e)] $P_{Y\mid X} (Y=3\mid X\leq 1)$
    \item [f)] $P_{X\mid Y} (1\leq X\leq 3 \mid 1\leq Y\leq 3)$
\end{itemize}
\end{multicols}

\begin{respostas}

\begin{itemize}
    \item [a)] 
    $P_{X\mid Y} (X=0\mid 1\leq Y\leq 3)
    = \sum_{y=1}^3 P_{XY}(0, y) / [F_Y(3) - F_Y(0)]
    = (3/32) / [1 - 4/32]
    = 3/28$.
    
    \item [b)] 
    $P_{Y\mid X} (Y=0\mid 1\leq X\leq 3)
    = \sum_{x=1}^3 P_{XY}(x, 0) / [F_X(3) - F_X(0)]
    = (3/32) / [1 - 4/32]
    = 3/28$.
    
    \item [c)] 
    $P_{Y\mid X} (Y=2\mid X\leq 2)
    = \sum_{x=0}^2 P_{XY}(x, 2) / F_X(2)
    = (13/32) / (28/32)
    = 13/28$.
    
    \item [d)] 
    $P_{X\mid Y} (X\leq 1\mid Y\geq 2)
    = \sum_{x=0}^1\sum_{y=2}^3 P_{XY}(x, y) / [1-F_Y(1)]
    = (7/32) / [1 - 14/32]
    = 7/18$.
    
    \item [e)] 
    $P_{Y\mid X} (Y=3\mid X\leq 1)
    = \sum_{x=0}^1 P_{XY}(x, 3) / F_X(1)
    = (2/32) / (14/32)
    = 2/14$.
    
    \item [f)] 
    $P_{X\mid Y} (1\leq X\leq 3\mid 1\leq Y\leq 3)
    = 1 - P_{X\mid Y} (X=0\mid 1\leq Y\leq 3)
    = 25/28$.
\end{itemize}

\end{respostas}
