\section{Questão 16}
Considere variáveis randômicas conjuntas $(X, Y)$.
Encontre a região do plano $xy$ correspondente aos eventos:
%
\begin{multicols}{2}
\begin{itemize}
    \item [a)] $A = \{(X,Y)\mid x + y \leq 2\}$
    \item [b)] $B = \{(X,Y)\mid x^2 + y^2 < 4\}$
    \item [c)] $C = \{(X,Y)\mid \min(x, y) \leq 2\}$
    \item [d)] $D = \{(X,Y)\mid \max(x, y) \leq 2\}$
\end{itemize}
\end{multicols}

\begin{respostas}
\begin{itemize}
    \item [a)] 
    A área abaixo de uma reta que passa por $(0, 2)$ e $(2, 0)$, incluindo a reta.
    
    \item [b)] 
    Um círculo de raio 2 centrado na origem, não incluindo a circunferência de raio 2 centrada na origem.
    
    \item [c)] 
    O plano $xy$ exceto o recorte retangular $x>2 \wedge y>2$.
    
    \item [d)] 
    O recorte retangular $x\leq 2 \wedge y\leq 2$.
\end{itemize}
\end{respostas}
