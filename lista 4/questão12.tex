\section{Questão 12}

Expresse $\E{X\mid Y}$ e $\E{Y\mid X}$ como variáveis randômicas cujo valores são representados por 
$g(y) = \E{X\mid Y = y}$ quando $Y = y$ e 
$g(x) = \E{Y\mid X = x}$ quando $X = x$, respectivamente.

\begin{respostas}
    Pelos resultados do item anterior,
    \begin{equation}
        g(z) = 
        \begin{cases}
            3/2,   & z=0 \vee z=1 \vee z=3 \\
            23/14, & z=2 \\
            0,     & z>2 \vee z<0
        \end{cases}.
    \end{equation}
\end{respostas}
