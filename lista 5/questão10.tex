\section{Questão 10}

Construa o diagrama de transição de estados, $S = \{1, 2, 3\}$, para a cadeia de Markov dada pela matriz de transição a seguir
%
$$
    P = \begin{bmatrix}
        0 & 1/2 & 1/2 \\
        1/4 & 3/4 & 0 \\
        1/4 & 3/4 & 0
    \end{bmatrix}
$$
\begin{respostas}
    \begin{figure}[htbp]
        \centering
        \begin{tikzpicture}[scale=2.25]
            \node[state] (1) {1};
            \node[state, right=2 of 1] (2) {2};
            \node[state, right=2 of 2] (3) {3};

            \path[->]
                (1) edge[bend left] node[below] {$1/2$} (2)
                (1) edge[bend left] node[above] {$1/2$} (3)
                (2) edge[bend left] node[above] {$1/4$} (1)
                (3) edge[bend left] node[below] {$1/4$} (1)
                (3) edge[bend left] node[above] {$3/4$} (2);
            \draw [->, thick, verdeunb] 
                (2.north) .. controls ++(60:0.3) and +(-60:-0.3) .. node[below right, black] {$3/4$} (2.north);
        \end{tikzpicture}
    \end{figure}
\end{respostas}
