\section{Questão 4}

Seja $X_0$, $X_1$, $X_2$, $\ldots$ uma sequência temporal de variáveis randômicas i.i.d. com média
$\E{X_i} = \frac{\pi}{2}$
e variância
$\Var{X_i} = \pi$.
Considere o processo aleatório em tempo discreto $\{Y[n], n \in \mathbb{N}\}$ definido como
%
$$
    Y[n] = \frac{1}{n} \sum_{i=0}^{n-1} X_i.
$$
Determine
%
\begin{multicols}{2}
\begin{itemize}
    \item [a)] $\mu_Y[n]$.
    \item [b)] $R_Y[m,n]$ para todo $m,n \in \mathbb{N}$.
\end{itemize}
\end{multicols}

\begin{respostas}
\begin{itemize}
    \item [a)] Pela linearidade da expectância, tem-se
    %
    \begin{equation}
        \mu_Y[n] 
        = \E{Y[n]}
        = \E{\frac{1}{n} \sum_{i=0}^{n-1} X_i}
        = \frac{1}{n} \sum_{i=0}^{n-1} \E{X_i}
        = \frac{1}{n} n \frac{\pi}{2} = \frac{\pi}{2}.
    \end{equation}

    \item [b)] Pela independência das variáveis aleatórias $X_i$, e assumindo $m \geq n$, tem-se
    %
    \begin{align}
        R_Y[m,n] 
        &= \E{Y[m]Y[n]} \\
        &= \E{\frac{1}{m} \sum_{i=0}^{m-1} X_i \cdot \frac{1}{n} \sum_{j=0}^{n-1} X_j} \\
        &= \frac{1}{mn} \sum_{i=0}^{m-1} \sum_{j=0}^{n-1} \E{X_i X_j} \\
        &= \frac{1}{mn} \sum_{i=0}^{m-1} \sum_{j=0}^{n-1} 
            \begin{cases}
                \Var{X_i} + \E{X_i}^2, & i = j \\
                \E{X_i}\E{X_j}, & i \neq j
            \end{cases} \\
        &= \frac{1}{mn} \left[
                n(\Var{X_i} + \E{X_i}^2) + n(m-1)\E{X_i}^2
            \right] \\
        &= \frac{\pi + \frac{\pi^2}{4}}{m} + \frac{m-1}{m}\frac{\pi^2}{4}\\
        &= \frac{\pi}{m} + \frac{\pi^2}{4}.
    \end{align}

\end{itemize}
\end{respostas}
