\section{Questão 14}

Considere a cadeia de Markov com três estados, $S = \{1, 2, 3\}$, que tem a seguinte matriz de transição
%
$$
    \begin{bmatrix}
        1/4 & 1/4 & 1/2 \\
        2/3 & 0   & 1/3 \\
        1/2 & 1/4 & 1/4
    \end{bmatrix}
$$
%
\begin{itemize}
    \item [a)] Construa o diagrama de transição de estado para esta cadeia de Markov.
    \item [b)] Se soubermos $P(X_0 = 1) = P(X_0 = 2) = 1/4$ encontre $P(X_0=3, X_1=2, X_2=1, X_3=1)$.
\end{itemize}

\begin{respostas}
    \begin{itemize}
        \item [a)] O diagrama de transição de estados é dado por
        %
        \begin{figure}[H]
            \centering
            \begin{tikzpicture}
                \node[state] (1) {1};
                \node[state, right=2 of 1] (2) {2};
                \node[state, right=2 of 2] (3) {3};

                \path [->]
                    (1) edge[bend left] node [below] {$1/4$} (2)
                    (1) edge[loop above] node [above] {$1/4$} (1)
                    (1) edge[bend left] node [above] {$1/2$} (3)
                    %
                    (2) edge[bend left] node [above] {$2/3$} (1)
                    (2) edge[bend left] node [below] {$1/3$} (3)
                    %
                    (3) edge[bend left] node [below] {$1/2$} (1)
                    (3) edge[bend left] node [above] {$1/4$} (2)
                    (3) edge[loop above] node [above] {$1/4$} (3);
            \end{tikzpicture}
        \end{figure}

        \item [b)] Tem-se
        \begin{align}
            P
            &= P(X_3 = 1 | X_2 = 1) \cdot 
               P(X_2 = 1 | X_1 = 2) \cdot 
               P(X_1 = 2 | X_0 = 3) \cdot 
               P(X_0 = 3) \\ 
            &= \frac{1}{3} \cdot \frac{1}{4} \cdot \frac{1}{4} \cdot 
               \left(1-\frac{1}{4}-\frac{1}{4}\right) \\
            &= \frac{1}{96}.
        \end{align}
    \end{itemize}
\end{respostas}
