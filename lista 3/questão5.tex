\section{Questão 5}
Considere a CDF de uma variável randômica contínua representada pela expressão matemática:
%
\begin{equation*}
    F_X(x) = 
    \begin{cases}
        0,     & x < 0 \\
        x + 1/2, & 0 \leq x < 1/2 \\
        1,     & x \geq 1/2
    \end{cases}
\end{equation*}

\begin{itemize}
    \item [a)] Esboce $F_X(x)$.
    \item [b)] Encontre a PDF de $X$.
    \item [c)] Determine $P(X > 1/4)$.
\end{itemize}

\respostas
\begin{itemize}
    \item [a), b), c)]
    Pela expressão da CDF, tem-se
    
    \begin{minipage}{.175\textwidth}
        \centering 
        \begin{tikzpicture}[> = latex, scale=1.25]
            \fill [azulunb!20, opacity=0.5]
                (0, 1/2) -- (0.5, 1) -- (1, 1) -- (1, 0) -- (0, 0) -- cycle
            ;
            \draw [->] (-.5, 0) -- (1.2, 0) node [right] {$x$};
            \draw [->] (0, 0) -- (0, 1.5) node [above] {$F_X(x)$};

            \fill (0, 0) circle (0.05) node [below] {$0$};
            \fill (0, 1/2) circle (0.05) node [left] {$\frac{1}{2}$};
            \draw [dashed] (1/2, 1) -- (1/2, 0) node [below] {$\frac{1}{2}$};
            \draw [dashed] (1/2, 1) -- (0, 1) node [left] {$1$};
            \draw [very thick, azulunb]
                (-0.5, 0) -- (0, 0)
                (0, 1/2) -- (0.5, 1) -- (1, 1)
            ;
        \end{tikzpicture}
    \end{minipage}
    %
    \begin{minipage}{.4\textwidth}
        \begin{equation}
            f_X(x) =
            \begin{cases}
                0,     & x < 0 \\
                1+\frac{1}{2}\delta(x),     & 0 \leq x < \frac12 \\
                0,     & x \geq \frac12
            \end{cases}
        \end{equation}
    \end{minipage}
    %
    \begin{minipage}{.35\textwidth}
        \begin{align}
            P\left(X > \frac{1}{4}\right) 
            %&= 1 - P(X < 1/4) \\
            &= 1 - F_X\left(\frac{1}{4}\right) \\
            &= 1 - \frac{3}{4} \\
            &= \frac{1}{4}
        \end{align}
    \end{minipage}
\end{itemize}
